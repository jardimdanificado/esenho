cmake_minimum_required(VERSION 3.22)
project(esenho_desktop C)
find_package(JNI REQUIRED)
set(CMAKE_C_STANDARD 11)
set(REPO ${CMAKE_CURRENT_SOURCE_DIR}/..)
set(ANDROID_CPP ${REPO}/android/app/src/main/cpp)
file(READ ${ANDROID_CPP}/esenho_jni.c JNI_SOURCE)
string(REPLACE "#include <android/bitmap.h>" "#include \"bitmap_compat.h\"" JNI_SOURCE "${JNI_SOURCE}")
string(REPLACE "#include <android/log.h>" "" JNI_SOURCE "${JNI_SOURCE}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/esenho_jni.c "${JNI_SOURCE}")
file(GLOB BRUSHES CONFIGURE_DEPENDS ${ANDROID_CPP}/gen/brushes/*.c)
file(GLOB FILTERS CONFIGURE_DEPENDS ${ANDROID_CPP}/gen/filters/*.c)
add_library(esenho_core SHARED ${CMAKE_CURRENT_BINARY_DIR}/esenho_jni.c ${ANDROID_CPP}/gen/esenho_plugins_registry.c ${BRUSHES} ${FILTERS} ${REPO}/src/quadro.c)
target_include_directories(esenho_core PRIVATE ${JNI_INCLUDE_DIRS} native ${ANDROID_CPP}/gen ${REPO}/src/include ${REPO}/src/plugins ${REPO}/src/plugins/brushes)
target_compile_options(esenho_core PRIVATE -O3 -w)
# GCC's -O3 vectorises the brush row loops badly: measured 4-33% fewer cycles
# for src/quadro.c at -O2 (circle h55 1/0.75, h100 1/0.92, square 1/0.96). The
# rest of the core keeps -O3. Clang, used by the Android NDK, is neutral here,
# so this only touches the desktop compiler.
set_source_files_properties(${REPO}/src/quadro.c PROPERTIES COMPILE_OPTIONS "-O2")
target_link_libraries(esenho_core m)
set(QJS ${ANDROID_CPP}/vendor/quickjs)
add_library(esenho_script SHARED ${ANDROID_CPP}/esenho_script_jni.c ${QJS}/quickjs.c ${QJS}/dtoa.c ${QJS}/libregexp.c ${QJS}/libunicode.c ${QJS}/cutils.c)
target_include_directories(esenho_script PRIVATE ${JNI_INCLUDE_DIRS} ${QJS})
target_compile_definitions(esenho_script PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(esenho_script PRIVATE -O2)
target_link_libraries(esenho_script m)
# Optional hardware valuators: AWT itself exposes only mouse coordinates.
find_package(X11)
if(X11_FOUND AND X11_Xi_FOUND)
    find_package(Threads REQUIRED)
    add_library(esenho_input SHARED native/tablet_input.c)
    target_include_directories(esenho_input PRIVATE ${JNI_INCLUDE_DIRS} ${X11_INCLUDE_DIR})
    target_link_libraries(esenho_input ${X11_LIBRARIES} ${X11_Xi_LIB} Threads::Threads m)
endif()
